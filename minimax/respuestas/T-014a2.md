import { t } from '@/lib/i18n';
import { cn } from '@/lib/utils';

/**
 * Estados visibles de una cita, incluyendo el calculado "en_curso".
 * Se declara aquí porque no existe en la base de datos.
 */
export type EstadoVisibleCita =
  | 'programada'
  | 'confirmada'
  | 'en_curso'
  | 'realizada'
  | 'cancelada'
  | 'no_asistida';

export type PuntoEstadoCitaProps = {
  estado: EstadoVisibleCita;
  conEtiqueta?: boolean;
};

/**
 * Punto de estado de la cita con su etiqueta opcional.
 * El punto es decorativo (aria-hidden) y la etiqueta sirve para lectores de pantalla.
 */
export function PuntoEstadoCita({ estado, conEtiqueta = true }: PuntoEstadoCitaProps) {
  const clasesPunto = cn(
    'inline-block w-2.5 h-2.5 rounded-full',
    {
      // Programada: hueco, contorno fino (1px)
      'border border-muted-foreground bg-transparent': estado === 'programada',
      // Confirmada: hueco, contorno reforzado (2px)
      'border-2 border-muted-foreground bg-transparent': estado === 'confirmada',
      // En curso: relleno con latido y halo
      'bg-primary latido-estado': estado === 'en_curso',
      // Realizada: relleno sin halo
      'bg-primary': estado === 'realizada',
      // Cancelada: hueco con raya diagonal
      'border border-destructive bg-transparent relative': estado === 'cancelada',
      // No asistida: partido (mitad rellena, mitad hueca)
      'border border-accent bg-[linear-gradient(90deg,var(--accent)_50%,transparent_50%)]':
        estado === 'no_asistida',
    }
  );

  // La etiqueta siempre va en text-foreground por contraste (nunca en color de punto)
  const clasesEtiqueta = 'text-foreground';

  return (
    <span className="flex items-center">
      {/* Punto decorativo, oculto para lectores de pantalla */}
      <span aria-hidden="true" className={clasesPunto}>
        {/* Raya diagonal para estado cancelada */}
        {estado === 'cancelada' && (
          <span className="absolute inset-x-[-1px] top-1/2 h-px -rotate-45 bg-destructive" />
        )}
      </span>
      {/* Etiqueta: visible por defecto, sr-only solo cuando conEtiqueta es false */}
      <span className={cn('ml-1', clasesEtiqueta, { 'sr-only': conEtiqueta === false })}>
        {t(`agenda.estados.${estado}`)}
      </span>
    </span>
  );
}

/**
 * Leyenda con los seis estados de la cita.
 * El punto de en_curso NO late aquí: solo late en la cita que realmente está en curso.
 */
export function LeyendaEstadosCita() {
  const estados: EstadoVisibleCita[] = [
    'programada',
    'confirmada',
    'en_curso',
    'realizada',
    'cancelada',
    'no_asistida',
  ];

  return (
    <ul aria-label={t('agenda.leyendaEstados')}>
      {estados.map((estado) => (
        <li key={estado} className="flex items-center mb-1">
          {/* Punto sin latido para la leyenda */}
          <PuntoEstadoCitaSinLatido estado={estado} />
          {/* Etiqueta visible */}
          <span className="ml-2 text-foreground">{t(`agenda.estados.${estado}`)}</span>
        </li>
      ))}
    </ul>
  );
}

/**
 * Variante del punto sin latido, usada solo en la leyenda.
 */
function PuntoEstadoCitaSinLatido({ estado }: { estado: EstadoVisibleCita }) {
  const clasesPunto = cn(
    'inline-block w-2.5 h-2.5 rounded-full',
    {
      // Programada: hueco, contorno fino (1px)
      'border border-muted-foreground bg-transparent': estado === 'programada',
      // Confirmada: hueco, contorno reforzado (2px)
      'border-2 border-muted-foreground bg-transparent': estado === 'confirmada',
      // En curso en leyenda: relleno pero SIN latido
      'bg-primary': estado === 'en_curso',
      // Realizada: relleno sin halo
      'bg-primary': estado === 'realizada',
      // Cancelada: hueco con raya diagonal
      'border border-destructive bg-transparent relative': estado === 'cancelada',
      // No asistida: partido (mitad rellena, mitad hueca)
      'border border-accent bg-[linear-gradient(90deg,var(--accent)_50%,transparent_50%)]':
        estado === 'no_asistida',
    }
  );

  return (
    <span className="flex items-center">
      <span aria-hidden="true" className={clasesPunto}>
        {estado === 'cancelada' && (
          <span className="absolute inset-x-[-1px] top-1/2 h-px -rotate-45 bg-destructive" />
        )}
      </span>
    </span>
  );
}

import { render, screen } from '@testing-library/react';
import { describe, expect, it } from 'vitest';
import { PuntoEstadoCita, LeyendaEstadosCita } from '@/components/ui/estadocita';
import { EstadoVisibleCita } from '@/components/ui/estadocita';

describe('PuntoEstadoCita', () => {
  // Prueba 1: cada estado muestra su etiqueta del catálogo
  it.each([
    'programada',
    'confirmada',
    'en_curso',
    'realizada',
    'cancelada',
    'no_asistida',
  ])('renderiza la etiqueta correcta para el estado %s', (estado) => {
    render(<PuntoEstadoCita estado={estado as EstadoVisibleCita} />);
    // Se usa el texto directamente del catálogo
    const textos: Record<string, string> = {
      programada: 'Programada',
      confirmada: 'Confirmada',
      en_curso: 'En curso',
      realizada: 'Realizada',
      cancelada: 'Cancelada',
      no_asistida: 'No asistió',
    };
    expect(screen.getByText(textos[estado])).toBeInTheDocument();
  });

  // Prueba 2: solo en_curso lleva la clase latido-estado
  it('solo el estado "en_curso" lleva la clase "latido-estado"', () => {
    const { container } = render(<PuntoEstadoCita estado="en_curso" />);
    expect(container.querySelector('.latido-estado')).toBeInTheDocument();

    const otrosEstados: EstadoVisibleCita[] = [
      'programada',
      'confirmada',
      'realizada',
      'cancelada',
      'no_asistida',
    ];
    otrosEstados.forEach((estado) => {
      const { container: c } = render(<PuntoEstadoCita estado={estado} />);
      expect(c.querySelector('.latido-estado')).not.toBeInTheDocument();
    });
  });

  // Prueba 3: conEtiqueta={false} pone sr-only
  it('conEtiqueta={false} mantiene la etiqueta en el DOM con "sr-only"', () => {
    render(<PuntoEstadoCita estado="programada" conEtiqueta={false} />);
    const etiqueta = screen.getByText('Programada');
    expect(etiqueta).toHaveClass('sr-only');
  });

  // Prueba 4: el punto lleva aria-hidden=true
  it('el punto lleva "aria-hidden=true"', () => {
    const { container } = render(<PuntoEstadoCita estado="programada" />);
    expect(container.querySelector('span[aria-hidden="true"]')).toBeInTheDocument();
  });
});

describe('LeyendaEstadosCita', () => {
  // Prueba 5: la leyenda tiene seis elementos y ninguno lleva latido-estado
  it('la leyenda tiene seis elementos de lista y ninguno lleva "latido-estado"', () => {
    render(<LeyendaEstadosCita />);
    const elementosLista = screen.getAllByRole('listitem');
    expect(elementosLista.length).toBe(6);
    elementosLista.forEach((elemento) => {
      expect(elemento.querySelector('.latido-estado')).not.toBeInTheDocument();
    });
  });

  // Prueba 6: los seis puntos se distinguen sin color
  it('los seis puntos se distinguen sin color', () => {
    render(<LeyendaEstadosCita />);
    // Seleccionamos los spans con aria-hidden="true" que son los puntos
    const puntos = screen.getAllByText(/Programada|Confirmada|En curso|Realizada|Cancelada|No asistió/)
      .map((el) => el.parentElement?.querySelector('span[aria-hidden="true"]'))
      .filter(Boolean) as HTMLElement[];

    // Extraemos clases quitando las de color
    const clasesForma = puntos.map((punto) => {
      const clases = punto.className.split(' ');
      // Quitar clases que contengan primary, accent, destructive o muted-foreground
      return clases
        .filter(
          (c) =>
            !c.includes('primary') &&
            !c.includes('accent') &&
            !c.includes('destructive') &&
            !c.includes('muted-foreground')
        )
        .sort()
        .join(' ');
    });

    // Debe haber 6 clases forma únicas
    const clasesUnicas = new Set(clasesForma);
    expect(clasesUnicas.size).toBe(6);
  });
});
