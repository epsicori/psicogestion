import { EstadoVisibleCita } from '@/lib/supabase/tipos-bd';
import { t } from '@/lib/i18n';
import { cn } from '@/lib/utils';

export type PuntoEstadoCitaProps = {
  estado: EstadoVisibleCita;
  conEtiqueta?: boolean;
};

export function PuntoEstadoCita({ estado, conEtiqueta = true }: PuntoEstadoCitaProps): JSX.Element {
  const clasesPunto = cn(
    'inline-block w-2.5 h-2.5 rounded-full',
    {
      'bg-primary': estado === 'en_curso' || estado === 'realizada',
      'bg-accent': estado === 'no_asistida',
      'bg-destructive': estado === 'cancelada',
      'bg-muted-foreground': estado === 'programada' || estado === 'confirmada',
      'border-2': estado === 'programada' || estado === 'confirmada',
      'border-primary': estado === 'en_curso',
      'border-dashed': estado === 'no_asistida',
      'line-through': estado === 'cancelada',
      'animate-latido-estado': estado === 'en_curso',
    },
    'mr-1'
  );

  const clasesEtiqueta = cn(
    'text-foreground',
    {
      'text-muted-foreground': estado === 'programada' || estado === 'confirmada',
      'text-destructive': estado === 'cancelada',
      'text-accent': estado === 'no_asistida',
    }
  );

  return (
    <span className="flex items-center">
      <span aria-hidden="true" className={clasesPunto}></span>
      {conEtiqueta !== false && (
        <span className={cn('sr-only', clasesEtiqueta)}>{t(`agenda.estados.${estado}`)}</span>
      )}
    </span>
  );
}

export function LeyendaEstadosCita(): JSX.Element {
  return (
    <ul aria-label={t('agenda.leyenda_estados')}>
      {[
        'programada',
        'confirmada',
        'en_curso',
        'realizada',
        'cancelada',
        'no_asistida',
      ].map((estado) => (
        <li key={estado} className="flex items-center mb-1">
          <PuntoEstadoCita estado={estado as EstadoVisibleCita} conEtiqueta={true} />
          <span className="ml-2 text-foreground">{t(`agenda.estados.${estado}`)}</span>
        </li>
      ))}
    </ul>
  );
}

import { render, screen } from '@testing-library/react';
import { PuntoEstadoCita, LeyendaEstadosCita } from '@/components/ui/estado-cita';
import { EstadoVisibleCita } from '@/lib/supabase/tipos-bd';

describe('PuntoEstadoCita', () => {
  it('renderiza la etiqueta correcta para cada estado', () => {
    // Prueba que cada estado muestra su etiqueta correspondiente del catálogo
    const estados: EstadoVisibleCita[] = [
      'programada',
      'confirmada',
      'en_curso',
      'realizada',
      'cancelada',
      'no_asistida',
    ];
    estados.forEach((estado) => {
      render(<PuntoEstadoCita estado={estado} />);
      expect(screen.getByText(t(`agenda.estados.${estado}`))).toBeInTheDocument();
    });
  });

  it('solo el estado "en_curso" lleva la clase "latido-estado"', () => {
    // Verifica que solo el estado "en_curso" tiene la clase de animación
    const { container } = render(<PuntoEstadoCita estado="en_curso" />);
    expect(container.querySelector('.latido-estado')).toBeInTheDocument();
    const otrosEstados = [
      'programada',
      'confirmada',
      'realizada',
      'cancelada',
      'no_asistida',
    ];
    otrosEstados.forEach((estado) => {
      const { container } = render(<PuntoEstadoCita estado={estado} />);
      expect(container.querySelector('.latido-estado')).not.toBeInTheDocument();
    });
  });

  it('conEtiqueta={false} mantiene la etiqueta en el DOM con "sr-only"', () => {
    // Verifica que la etiqueta sigue existiendo con "sr-only" cuando conEtiqueta es false
    render(<PuntoEstadoCita estado="programada" conEtiqueta={false} />);
    expect(screen.getByText(t('agenda.estados.programada')).className).toContain('sr-only');
  });

  it('el punto lleva "aria-hidden=true"', () => {
    // Verifica que el punto tiene el atributo "aria-hidden" establecido en true
    const { container } = render(<PuntoEstadoCita estado="programada" />);
    expect(container.querySelector('span[aria-hidden="true"]')).toBeInTheDocument();
  });
});

describe('LeyendaEstadosCita', () => {
  it('la leyenda tiene seis elementos de lista y ninguno lleva "latido-estado"', () => {
    // Verifica que la leyenda tiene seis elementos de lista y ninguno tiene la clase de animación
    render(<LeyendaEstadosCita />);
    const elementosLista = screen.getAllByRole('listitem');
    expect(elementosLista.length).toBe(6);
    elementosLista.forEach((elemento) => {
      expect(elemento.querySelector('.latido-estado')).not.toBeInTheDocument();
    });
  });

  it('los seis puntos se distinguen sin color', () => {
    // Verifica que los seis puntos tienen diferentes clases de forma sin depender del color
    render(<LeyendaEstadosCita />);
    const puntos = screen.getAllByRole('img');
    const clasesForma = puntos.map((punto) => punto.className);
    const clasesUnicas = new Set(clasesForma);
    expect(clasesUnicas.size).toBe(6);
  });
});
