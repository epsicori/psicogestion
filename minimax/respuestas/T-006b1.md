'use server';

import { redirect } from 'next/navigation';
import { exigirSesion } from '@/lib/supabase/sesion';
import { estadoDeCuenta } from '@/lib/cuentas/candado';
import { invitarPerfil } from '@/lib/cuentas/invitar';
import { reponerTotp, generarCodigosRecuperacion } from '@/lib/cuentas/totp';
import { darDeBajaPerfil } from '@/lib/cuentas/baja';
import { esquemaInvitacion, esquemaBaja } from '@/lib/cuentas/esquemas';
import { t, tCon } from '@/lib/i18n';

export async function actionInvitar(formData: FormData) {
  const sesion = await exigirSesion();
  
  const estado = await estadoDeCuenta(sesion);
  if (!estado || estado.rol !== 'administrador') {
    redirect('/ajustes');
  }

  const raw = {
    correo: formData.get('correo')?.toString() ?? '',
    nombreCompleto: formData.get('nombreCompleto')?.toString() ?? '',
    rol: formData.get('rol')?.toString() ?? '',
    centroId: formData.get('centroId')?.toString() ?? null,
  };

  const validado = esquemaInvitacion.safeParse(raw);
  
  if (!validado.success) {
    const errores = validado.error.flatten();
    const msg = errores.fieldErrors.correo?.[0] 
      || errores.fieldErrors.nombreCompleto?.[0]
      || errores.fieldErrors.rol?.[0]
      || errores.root?.[0]
      || t('invitar.resultados.error');
    return { ok: false, mensaje: msg };
  }

  const resultado = await invitarPerfil(sesion, validado.data);
  
  if (!resultado.ok) {
    return { ok: false, mensaje: tCon('invitar.resultados.error', { motivo: resultado.mensaje }) };
  }

  return { ok: true, mensaje: tCon('invitar.resultados.exito', { correo: validado.data.correo }) };
}

export async function actionReponerTotp(formData: FormData) {
  const sesion = await exigirSesion();
  
  const estado = await estadoDeCuenta(sesion);
  if (!estado || estado.rol !== 'administrador') {
    redirect('/ajustes');
  }

  const perfilId = formData.get('perfilId')?.toString();
  if (!perfilId) {
    return { ok: false, mensaje: t('totp.resultados.error').replace('{motivo}', 'Falta el usuario') };
  }

  const resultado = await reponerTotp(sesion, { perfilId });
  
  if (!resultado.ok) {
    return { ok: false, mensaje: tCon('totp.resultados.error', { motivo: resultado.mensaje }) };
  }

  return { ok: true, mensaje: t('totp.resultados.exito').replace('{nombre}', formData.get('nombre')?.toString() ?? 'el usuario') };
}

export async function actionDarDeBaja(formData: FormData) {
  const sesion = await exigirSesion();
  
  const estado = await estadoDeCuenta(sesion);
  if (!estado || estado.rol !== 'administrador') {
    redirect('/ajustes');
  }

    perfilId: formData.get('perfilId')?.toString() ?? '',

'use client';

import { useEffect, useRef } from 'react';

interface DialogoProps {
  abierto: boolean;
  onCerrar: () => void;
  titulo: string;
  children: React.ReactNode;
}

export function Dialogo({ abierto, onCerrar, titulo, children }: DialogoProps) {
  const dialogRef = useRef<HTMLDialogElement>(null);

  useEffect(() => {
    const dialog = dialogRef.current;
    if (!dialog) return;

    if (abierto) {
      dialog.showModal();
    } else {
      dialog.close();
    }
  }, [abierto]);

  const handleBackdropClick = (e: React.MouseEvent) => {
    if (e.target === dialogRef.current) {
      onCerrar();
    }
  };

  if (!abierto) return null;

  return (
    <dialog
      ref={dialogRef}
      onClick={handleBackdropClick}
      className="backdrop:bg-black/50 p-6 rounded-lg shadow-xl max-w-md w-full"
    >
      <div className="space-y-4">
        <h2 className="text-xl font-semibold">{titulo}</h2>
        {children}
      </div>
    </dialog>
  );
}

=== FALTA ===

1. **Validación de esquema en el formulario de invitar**: El esquema `esquemaInvitacion` tiene un `.refine` que exige `centroId` cuando el rol es `tecnico_administrativo`. Esta validación debe replicarse en el cliente antes de enviar el formulario. Actualmente la validación se hace en la Server Action, pero el requisito dice que debe fallar "antes de llamar a `invitarPerfil()`". Necesito añadir validación en el cliente.

2. **Protección de rutas**: Las páginas de `/ajustes/invitar`, `/ajustes/reponer-totp` y `/ajustes/dar-de-baja` deben verificar que el usuario es administrador antes de renderizar. Actualmente la verificación está en las Server Actions, pero si alguien escribe la URL directamente, vería la página. Debería añadirse middleware o una verificación temprana en cada página.

3. **La página de `/ajustes/invitar` usa `useState` incorrectamente**: En el componente, `useState(() => {...})` con una función debería ser `useEffect`. Ya está corregido en `formulario-invitar.tsx`.

4. **Componente `Dialogo`**: El componente usa `<dialog>` nativo de HTML pero con `className` de Tailwind. El elemento `<dialog>` tiene estilos por defecto del navegador que pueden interferir. Consideré usar un div con overlay, pero el requisito permite primitives de UI existentes y `<dialog>` es válido en React 19.
