import { Cargando } from '@/components/ui/piezas';

import { Field } from '@/components/ui/field';
import { Dialogo } from '@/components/ui/dialogo';
import { t } from '@/lib/i18n';
import { exigirSesion, estadoDeCuenta, desbloquearHistoria, bloquearHistoria } from '@/lib/cuentas/candado';
import { z } from 'zod';

const esquemaPin = z.object({
  pin: z.string().regex(/^\d{6}$/).describe('PIN de seis dígitos'),
});

async function ContenidoCandado() {
  const sesion = await exigirSesion();
  const estadoCuenta = await estadoDeCuenta(sesion);

  if (!estadoCuenta || !estadoCuenta.requierePin) {
    // Redirigir a otra página si el candado no es necesario
    return null;
  }

  const { desbloquearHistoria, bloquearHistoria } = sesion;

  const [pin, setPin] = useState('');
  const [mensaje, setMensaje] = useState('');
  const [mostrarDialogo, setMostrarDialogo] = useState(false);

  const enviarPin = async () => {
    try {
      const resultado = await desbloquearHistoria(sesion, pin);
      if (resultado.desbloqueado) {
        setMensaje(t('candado.desbloqueado'));
      } else {
        setMensaje(t('candado.bloqueado'));
      }
    } catch (error) {
      setMensaje(t('candado.error'));
    }
  };

  const cerrarSesion = async () => {
    await bloquearHistoria(sesion);
    setMensaje(t('candado.sesion_bloqueada'));
  };

  return (
    <main>
      <header>
        <h1>{t('candado.titulo')}</h1>
      </header>
      <section>
        <Field
          id="pin"
          etiqueta={t('candado.pin')}
          errores={[]}
          {...register('pin')}
          value={pin}
          onChange={(e) => setPin(e.target.value)}
        />
        <Button variante="primario" onClick={enviarPin}>
          {t('candado.boton_desbloquear')}
        </Button>
        <Button variante="secundario" onClick={cerrarSesion}>
          {t('candado.boton_bloquear')}
        </Button>
        {mensaje && <p>{mensaje}</p>}
      </section>
      {mostrarDialogo && (
        <Dialogo
          titulo={t('candado.dialogo_titulo')}
          mensaje={t('candado.dialogo_mensaje')}
          botones={[
            { texto: t('candado.dialogo_boton_ok'), accion: () => setMostrarDialogo(false) },
          ]}
          onClose={() => setMostrarDialogo(false)}
        />
      )}
    </main>
  );
}

export default function PaginaCandado() {
  return (
    <main>
      <header>
        <h1>{t('candado.titulo')}</h1>
      </header>
      <Suspense fallback={<Cargando filas={3} />}>
        <ContenidoCandado />
      </Suspense>
    </main>
  );
}
