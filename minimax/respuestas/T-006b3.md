// Componente de temporizador para bloqueo
function CountdownTimer({ fechaFin }: { fechaFin: string }) {
  const [restante, setRestante] = useState('');

  useEffect(() => {
    const actualizar = () => {
      const ahora = new Date().getTime();
      const bloqueo = new Date(fechaFin).getTime();
      const diff = bloqueo - ahora;

      if (diff <= 0) {
        setRestante('0:00');
        return;
      }

      const minutos = Math.floor(diff / 60000);
      const segundos = Math.floor((diff % 60000) / 1000);
      setRestante(`${minutos}:${segundos.toString().padStart(2, '0')}`);
    };

    actualizar();
    const intervalo = setInterval(actualizar, 1000);
    return () => clearInterval(intervalo);
  }, [fechaFin]);

  return (
    <div className="text-center">
      <div className="text-4xl font-mono font-bold text-error">
        {restante}
      </div>
    </div>
  );
}

// Componente de aviso de caducidad (no modal, no roba foco)
function AvisoCaducidad({ caducaEn }: { caducaEn: string }) {
  const [visible, setVisible] = useState(false);
  const [tiempoRestante, setTiempoRestante] = useState('');

  useEffect(() => {
    const actualizar = () => {
      const ahora = new Date().getTime();
      const caduca = new Date(caducaEn).getTime();
      const diff = caduca - ahora;

      // Solo mostrar cuando queden menos de 30 segundos (tarde)
      if (diff <= 30000 && diff > 0) {
        setVisible(true);
        const segundos = Math.floor(diff / 1000);
        setTiempoRestante(`${segundos}s`);
      } else {
        setVisible(false);
      }
    };

    actualizar();
    const intervalo = setInterval(actualizar, 1000);
    return () => clearInterval(intervalo);
  }, [caducaEn]);

  if (!visible) return null;

  return (
    <div className="mt-4 p-3 bg-warning/10 border border-warning/30 rounded text-sm text-warning">
      {t('candado.avisoCaducidad', { tiempo: tiempoRestante })}
    </div>
  );
}

'use server';

import { exigirSesion } from '@/lib/supabase/sesion';
import {
  estadoDeCuenta,
  desbloquearHistoria,
  bloquearHistoria,
  prolongarDesbloqueo,
} from '@/lib/cuentas/candado';
import { t } from '@/lib/i18n';

interface ResultadoDesbloqueo {
  desbloqueado: boolean;
  motivo: string;
  caducaEn: string | null;
  bloqueadoHasta: string | null;
}

export async function desbloqHistoria(pin: string | null): Promise<ResultadoDesbloqueo> {
  const sesion = await exigirSesion();

  // Verificar estado de la cuenta
  const estado = await estadoDeCuenta(sesion);
  if (!estado) {
    return {
      desbloqueado: false,
      motivo: 'cuenta_inexistente',
      caducaEn: null,
      bloqueadoHasta: null,
    };
  }

  // Si no requiere PIN, está desbloqueado
  if (!estado.requierePin || !estado.tienePin) {
    return {
      desbloqueado: true,
      motivo: 'sin_pin',
      caducaEn: null,
      bloqueadoHasta: null,
    };
  }

  // Si ya está desbloqueado, devolver estado actual
  if (estado.desbloqueoCaducaEn) {
    const caduca = new Date(estado.desbloqueoCaducaEn);
    const ahora = new Date();
    
    if (caduca > ahora) {
      return {
        desbloqueado: true,
        motivo: 'ya_desbloqueado',
        caducaEn: estado.desbloqueoCaducaEn,
        bloqueadoHasta: null,
      };
    }
  }

  // Si no hay PIN proporcionado, es la primera llamada (para saber estado)
  if (pin === null) {
    return {
      desbloqueado: false,
      motivo: 'pendiente',
      caducaEn: null,
      bloqueadoHasta: null,
    };
  }

  // Intentar desbloquear con el PIN
  const resultado = await desbloquearHistoria(sesion, pin);

  // IMPORTANTE: La base iguala "PIN incorrecto" con "ya bloqueado" a propósito
  // La pantalla NO debe distinguir entre ambos casos
  return {
    desbloqueado: resultado.desbloqueado,
    motivo: resultado.motivo,
    caducaEn: resultado.caducaEn,
    bloqueadoHasta: resultado.bloqueadoHasta,
  };
}

export async function bloqHistoria(): Promise<number> {
  const sesion = await exigirSesion();
  const cerradas = await bloquearHistoria(sesion);
  return cerradas;
}

export async function prolongarDesbloq(): Promise<string | null> {
  const sesion = await exigirSesion();
  const nuevaCaducidad = await prolongarDesbloqueo(sesion);
  return nuevaCaducidad;
}

function CountdownTimer({ fechaFin }: { fechaFin: string }) {
  const [restante, setRestante] = useState('');

  useEffect(() => {
    const actualizar = () => {
      const ahora = new Date().getTime();
      const bloqueo = new Date(fechaFin).getTime();
      const diff = bloqueo - ahora;

      if (diff <= 0) {
        setRestante('0:00');
        return;
      }

      const minutos = Math.floor(diff / 60000);
      const segundos = Math.floor((diff % 60000) / 1000);
      setRestante(`${minutos}:${segundos.toString().padStart(2, '0')}`);
    };

    actualizar();
    const intervalo = setInterval(actualizar, 1000);
    return () => clearInterval(intervalo);
  }, [fechaFin]);

  return (
    <div className="text-center">
      <div className="text-4xl font-mono font-bold text-error">
        {restante}
      </div>
    </div>
  );
}

  // Mensaje genérico para todos

    canjeado: false,
