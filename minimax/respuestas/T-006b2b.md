    redirect('/ajustes');
  }

  // Si requiere PIN, ese es el primer paso

    redirect('/primera-vez/pin');

'use server';

import { redirect } from 'next/navigation';
import { exigirSesion } from '@/lib/supabase/sesion';
import { esquemaPin } from './esquemas';
import { estadoDeCuenta, fijarPinHistoria } from '@/lib/countas/candado';
import { t } from '@/lib/i18n';

type EstadoFormulario = {
  errores?: Record<string, string[]>;
  mensaje?: string;
};

export async function establecerPin(datos: unknown): Promise<EstadoFormulario> {
  const sesion = await exigirSesion();

  // Verificar que el usuario tiene acceso a este paso
  const estado = await estadoDeCuenta(sesion);
  if (!estado || !estado.requierePin) {
    return { mensaje: t('acceso.sinPermiso') };
  }

  const resultado = esquemaPin.safeParse(datos);

  if (!resultado.success) {
    const errores = resultado.error.flatten().fieldErrors;
    return { errores };
  }

  const { error } = await fixerPinHistoria(sesion, resultado.data.pin);

  if (error) {
    return { mensaje: t('acceso.errorEstablecerPin') };
  }

  redirect('/ajustes');
}

'use client';

import { useState } from 'react';
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { esquemaPinFormulario, type DatosPinFormulario } from './esquemas';
import { establecerPin } from './acciones';
import { Button } from '@/components/ui/button';
import { Field } from '@/components/ui/field';
import { ESTADO_INICIAL, aplicarErroresDelServidor } from '@/lib/formularios';

export function FormularioPin() {
  const [enviando, setEnviando] = useState(false);
  const [mensaje, setMensaje] = useState('');

  const {
    register,
    handleSubmit,
    setError,
    formState: { errors },
  } = useForm<DatosPinFormulario>({
    resolver: zodResolver(esquemaPinFormulario),
    defaultValues: {
      pin: '',
      confirmarPin: '',
    },
  });

  const onSubmit = handleSubmit(async (datos) => {
    setEnviando(true);
    setMensaje('');

    const estado = await establecerPin(datos);

    if (estado.errores || estado.mensaje) {
      aplicarErroresDelServidor(estado, setError);
      setMensaje(estado.mensaje ?? '');
      setEnviando(false);
    }
    // Si todo va bien, la acción hace redirect
  });

  return (
    <form onSubmit={onSubmit} className="space-y-4">
      <Field
        id="pin"
        type="password"
        label="PIN"
        placeholder="123456"
        maxLength={6}
        errores={errors.pin?.message ? [errors.pin.message] : []}
        {...register('pin')}
      />

      <Field
        id="confirmarPin"
        type="password"
        label="Confirmar PIN"
        placeholder="123456"
        maxLength={6}
        errores={errors.confirmarPin?.message ? [errors.confirmarPin.message] : []}
        {...register('confirmarPin')}
      />

      {mensaje && (
        <p className="text-sm text-red-600" role="alert">
          {mensaje}
        </p>
      )}

      <Button
        type="submit"
        variante="primario"
        tamano="md"
        className="w-full"
        disabled={enviando}
      >
        {enviando ? 'Estableciendo...' : 'Establecer PIN'}
      </Button>
    </form>
  );
}

import { redirect } from 'next/navigation';
import { exigesion } from '@/lib/supabase/sesion';
import { estadoDeCuenta } from '@/lib/cuentas/candado';

export default async function LayoutGrupoSesion({
  children,
}: {
  children: React.ReactNode;
}) {
  const sesion = await exigesion();

  // Obtener el nivel de aseguramiento de la sesión
  const { data: nivelAuth } = await sesion.supabase.auth.mfa.getAuthenticatorAssuranceLevel();

  // Obtener el estado de la cuenta desde la base de datos
  const estado = await estadoDeCuenta(sesion);

  // Si no hay estado de cuenta, algo está mal con el perfil
  if (!estado) {
    redirect('/login');
  }

  // Verificar si la sesión actual tiene TOTP verificado (aal2)
  const tieneTotp = nivelAuth?.currentLevel === 'aal2';

  // Si la sesión no tiene TOTP verificado, necesita completar ese paso
  // (la verificación de PIN se hace en primera-vez)
  if (!tieneTotp) {
    // Redirigir a la página de primera vez para determinar el paso pendiente
    redirect('/primera-vez');
  }

  // Si requiere PIN (y no es técnico administrativo), necesita establecerlo
  if (estado.requierePin && estado.rol !== 'tecnico_administrativo') {
    redirect('/primera-vez/pin');
  }

  // Si todo está completo, renderizar los hijos
  return <>{children}</>;
}
