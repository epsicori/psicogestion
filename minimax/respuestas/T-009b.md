on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

  build:
    runs-on: ubuntu-latest

    services:

        image: supabase/supabase:latest
        ports:
          - 5432:5432
        env:

          --health-interval 10s
          --health-timeout 5s
          --health-retries 5

    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Set up Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '18'
          cache: 'npm'

      - name: Install dependencies
        run: npm install

      - name: Apply Supabase migrations
        run: npx supabase migration apply

      - name: Generate Supabase types
        run: npm run tipos

      - name: Check for type changes
        run: |
          if [[ $(git diff --name-only HEAD HEAD^) == *"lib/supabase/tipos-bd.ts"* ]]; then
            echo "Type changes detected. Failing the build."
            exit 1
          fi

      - name: Run lint
        run: npm run lint

      - name: Run build
        run: npm run build

      - name: Run tests
        run: npm test

      - name: Run RLS tests
        run: npm run test:rls

      - name: Run huella verifier
        run: npm run verificar:huellas

      - name: Run accessibility tests
        run: |
          # Install Playwright
          npm install playwright --save-dev
          npx playwright install --with-deps
          
          # Run accessibility tests
          npx playwright test --project=chromium
        env:
          SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
          SUPABASE_ANON_KEY: ${{ secrets.SUPABASE_ANON_KEY }}

  # Job to run migrations incrementally
  migrate:
    runs-on: ubuntu-latest
    needs: build
    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Set up Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '18'
          cache: 'npm'

      - name: Install dependencies
        run: npm install

      - name: Apply Supabase migrations incrementally
        run: npx supabase migration apply --name <migration-name>

      - name: Run tests after migration
        run: npm test

  # Job to reset database and apply migrations from scratch
  db-reset:
    runs-on: ubuntu-latest
    needs: build
    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Set up Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '18'
          cache: 'npm'

      - name: Install dependencies
        run: npm install

      - name: Reset Supabase database
        run: npx supabase db reset

      - name: Apply Supabase migrations
        run: npx supabase migration apply

      - name: Run tests after reset
        run: npm test

### Explicación de los fallos específicos:

1. **Flujo completo en verde**: El flujo pasa si todas las etapas (lint, build, test, test:rls, verificar:huellas, accesibilidad) se completan sin errores. El registro nombrará cada comprobación.

2. **Política de RLS rota**: Si la política de RLS está rota, el comando `npm run test:rls` fallará y el flujo se detendrá antes de la construcción.

3. **Tabla de solo adición sin capas**: Si una tabla de solo adición no tiene sus tres capas, el verificador de cobertura fallará porque el módulo `11-cobertura.sql` no estará cubierto adecuadamente.

4. **Edición manual de tipos**: Si `lib/supabase/tipos-bd.ts` se edita manualmente, el flujo detectará cambios en el archivo y fallará debido a la deriva de tipos.

5. **Contraste por debajo de AA**: Si el contraste de un token se baja por debajo de AA, el verificador de accesibilidad fallará y el flujo se detendrá.
