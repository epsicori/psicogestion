```tsx
import { t, tCon } from '@/lib/i18n';
import { Pildora } from '@/components/ui/piezas';

const estadosDeBloqueo: readonly string[] = ['festivo', 'ausencia', 'ocupada'];

export function CuerpoSesionPrevista({
  fecha,
  hora,
  estado,
  alternativa,
  horaResultante,
}: {
  fecha: string;
  hora: string | null;
  estado: 'normal' | 'festivo' | 'ausencia' | 'ocupada' | 'cambioDeHora';
  alternativa: string | null;
  horaResultante: string | null;
}) {
  const esBloqueo = estadosDeBloqueo.includes(estado);

  return (
    <div>
      <span className="text-sm font-medium text-foreground">{fecha}</span>

      {!esBloqueo && hora && (
        <span className="ml-2 font-mono text-sm text-foreground">{hora}</span>
      )}

      {estado === 'festivo' && (
        <>
          <Pildora tono="warning">{t('agenda.alta.serie.festivo')}</Pildora>
          {alternativa !== null ? (
            <span className="text-sm text-muted-foreground">
              {tCon('agenda.alta.serie.alternativa', { cuando: alternativa })}
            </span>
          ) : (
            <span className="text-sm text-muted-foreground">
              {t('agenda.alta.serie.sinAlternativa')}
            </span>
          )}
        </>
      )}

      {estado === 'ausencia' && (
        <>
          <Pildora tono="warning">{t('agenda.alta.serie.ausencia')}</Pildora>
          {alternativa !== null ? (
            <span className="text-sm text-muted-foreground">
              {tCon('agenda.alta.serie.alternativa', { cuando: alternativa })}
            </span>
          ) : (
            <span className="text-sm text-muted-foreground">
              {t('agenda.alta.serie.sinAlternativa')}
            </span>
          )}
        </>
      )}

      {estado === 'ocupada' && (
        <>
          <Pildora tono="warning">{t('agenda.alta.serie.ocupada')}</Pildora>
          {alternativa !== null ? (
            <span className="text-sm text-muted-foreground">
              {tCon('agenda.alta.serie.alternativa', { cuando: alternativa })}
            </span>
          ) : (
            <span className="text-sm text-muted-foreground">
              {t('agenda.alta.serie.sinAlternativa')}
            </span>
          )}
        </>
      )}

      {estado === 'cambioDeHora' && (
        <Pildora tono="info">
          {tCon('agenda.alta.serie.cambioDeHora', { hora: horaResultante ?? '' })}
        </Pildora>
      )}

      {estado === 'normal' && null}
    </div>
  );
}
```
