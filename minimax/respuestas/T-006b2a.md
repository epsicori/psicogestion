// tipo que verifica que una clave existe en el catálogo
export type ClaveDeCatalogo = typeof cadenas;

// función de traducción con interpolación
export function t(clave: string, params?: Record<string, string | number>): string {
  const partes = clave.split('.');
  let valor: unknown = cadenas;

  for (const parte of partes) {
    if (valor && typeof valor === 'object' && parte in valor) {
      valor = (valor as Record<string, unknown>)[parte];
    } else {
      console.error(`[i18n] Clave no encontrada: ${clave}`);
      return clave;
    }
  }

  if (typeof valor !== 'string') {
    console.error(`[i18n] La clave ${clave} no es una cadena`);
    return clave;
  }

  if (params) {
    return Object.entries(params).reduce(
      (acc, [k, v]) => acc.replace(`{${k}}`, String(v)),
      valor
    );
  }

  return valor;
}

// versión con contexto para mensajes de acción
export function tCon(clave: string, params?: Record<string, string | number>): string {
  return t(clave, params);
}

import { useRouter } from 'next/navigation';

  codigo: string;
};

const esquemaVerificacion = z.object({
  codigo: z.string().length(6, 'Código de 6 dígitos'),
});

type FactorAuth = {
  id: string;
  friendlyName?: string;
  secret?: string;
  qrCode?: string;

  const router = useRouter();
  const [supabase] = useState(() => 
    createBrowserClient<Database>(
      process.env.NEXT_PUBLIC_SUPABASE_URL!,
      process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
    )
  );
  
  const [paso, setPaso] = useState<Paso>('enroll');
  const [factorId, setFactorId] = useState<string | null>(null);
  const [challengeId, setChallengeId] = useState<string | null>(null);
  const [qrCode, setQrCode] = useState<string | null>(null);
  const [secret, setSecret] = useState<string | null>(null);
  const [codigosRecuperacion, setCodigosRecuperacion] = useState<string[]>([]);
  const [error, setError] = useState<string>('');
  const [cargando, setCargando] = useState(false);

  const {
    register,
    handleSubmit,
    formState: { errors },

  } = useForm<DatosVerificacion>({
    resolver: zodResolver(esquemaVerificacion),
    defaultValues: {
      codigo: '',
    },
  });

  // Enrolar TOTP al montar el componente
  useEffect(() => {
    async function enrolar() {
      try {
        const { data, error: errorEnroll } = await supabase.auth.mfa.enroll({
          factorType: 'totp',
          friendlyName: 'Psicogestión',
        });

        if (errorEnroll || !data) {
          console.error('[primera-vez] Error al enrolar TOTP:', errorEnroll);
          setError(t('primeraVez.errores.enrolar'));
          return;
        }

          // El QR viene como SVG, hay que anteponer el prefijo data URI

          setSecret(data.totp.secret);
        }

        // Iniciar el challenge automáticamente
        const { data: reto, error: errorChallenge } = await supabase.auth.mfa.challenge({
          factorId: data.id,
        });

        if (errorChallenge || !reto) {
          console.error('[primera-vez] Error al iniciar challenge:', errorChallenge);
          setError(t('primeraVez.errores.enrolar'));
          return;
        }

        setChallengeId(reto.id);
        setPaso('verify');
      } catch (err) {
        console.error('[primera-vez] Excepción en enrolar:', err);
        setError(t('primeraVez.errores.enrolar'));
      }
    }

    enrolar();
  }, [supabase]);

  const onVerificar = async (datos: DatosVerificacion) => {
    if (!factorId || !challengeId) {
      setError(t('primeraVez.errores.enrolar'));
      return;
    }

    setCargando(true);
    setError('');

    try {
      const { error: errorVerify } = await supabase.auth.mfa.verify({
        factorId,
        challengeId,
        code: datos.codigo,
      });

      if (errorVerify) {
        setError(t('primeraVez.pasoTotp.errorVerificacion'));
        setCargando(false);
        
        // Reiniciar el challenge para el siguiente intento
        const { data: nuevoReto } = await supabase.auth.mfa.challenge({ factorId });
        if (nuevoReto) {
          setChallengeId(nuevoReto.id);
        }
        return;

      // Verificación exitosa, generar códigos de recuperación
      const { data: sessionData } = await supabase.auth.getSession();
      if (!sessionData.session) {
        setError(t('sesion.expirada'));
        setCargando(false);
        return;
      }

      const response = await fetch('/primera-vez/acciones', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ 
          accion: 'generarCodigos', 
          userId: sessionData.session.user.id 
        }),

        setError(t('primeraVez.errores.generarCodigos'));
        setCargando(false);
        return;
      }

      setCodigosRecuperacion(resultado.codigos);
      setPaso('codigos');
      setCargando(false);
    } catch (err) {
      console.error('[primera-vez] Error en verificación:', err);
      setError(t('primeraVez.pasoTotp.errorGenerico'));
      setCargando(false);
    }
  };

  const onContinuar = () => {
    setPaso('completado');
    router.push('/');
  };

  const copiarCodigos = () => {
    navigator.clipboard.writeText(codigosRecuperacion.join('\n'));
  };

  // Paso: Enrolamiento (cargando)
  if (paso === 'enroll') {
    return (
      <main className="min-h-screen flex items-center justify-center bg-fondo">
        <div className="max-w-md w-full p-6">
          <h1 className="text-2xl font-bold text-center mb-4">
            {t('primeraVez.titulo')}
          </h1>
          <p className="text-center text-gray-600 mb-8">
            {t('primeraVez.subtitulo')}
          </p>
          <Cargando filas={5} />
        </div>
      </main>
    );
  }

  // Paso: Verificación TOTP

    return (
      <main className="min-h-screen flex items-center justify-center bg-fondo p-4">
        <div className="max-w-md w-full p-6 bg-white rounded-lg shadow-md">

            {t('primeraVez.pasoTotp.titulo')}
          </h1>
          <p className="text-center text-gray-600 mb-6">
            {t('primeraVez.pasoTotp.descripcion')}
          </p>

          {qrCode && (
            <div className="flex justify-center mb-6">
              <img 
                src={qrCode}

          <details className="mb-6">
            <summary className="cursor-pointer text-sm text-gray-500 hover:text-gray-700">
              {t('primeraVez.pasoTotp.noEscanear')}
            </summary>
            <div className="mt-2 p-3 bg-gray-50 rounded">
              <p className="text-xs text-gray-500 mb-1">
                {t('primeraVez.pasoTotp.secretoManual')}

'use server';

import { headers } from 'next/headers';
import { redirect } from 'next/navigation';
import { crearClienteNavegador } from '@/lib/supabase/navegador';
import { obtenerUsuario } from '@/lib/supabase/sesion';
import { generarCodigosRecuperacion } from '@/lib/cuentas/totp';
import { t } from '@/lib/i18n';

type AccionGenerarCodigos = {
  accion: 'generarCodigos';
  userId: string;
};

type Accion = AccionGenerarCodigos;

type EstadoFormulario = {
  ok?: boolean;
  errores?: Record<string, string[]>;
  mensaje?: string;
};

export async function accionPrimeriaVez(datos: unknown): Promise<EstadoFormulario> {
  const headersList = headers();
  const cookie = headersList.get('cookie') || '';
  
  const supabase = crearClienteNavegador(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    cookie
  );

  const { usuario } = await obtenerUsuario();

  if (!usuario) {
    return {
      ok: false,
      mensaje: t('sesion.expirada'),
    };
  }

  const entrada = datos as Accion;

  if (entrada.accion === 'generarCodigos') {
    try {
      const resultado = await generarCodigosRecuperacion(
        { supabase, usuario },
        usuario.id
      );

      if (!resultado.ok) {
        return {
          ok: false,
          mensaje: resultado.mensaje,
        };
      }

      return {
        ok: true,
        mensaje: '',
        // @ts-ignore - los códigos se devuelven en el objeto ok
        codigos: resultado.codigos,
      };
    } catch (error) {
      console.error('[primera-vez/acciones] Error al generar códigos:', error);
      return {
        ok: false,
        mensaje: t('primeraVez.errores.generarCodigos'),
      };
    }
  }

  return {
    ok: false,
    mensaje: t('comun.error'),
  };
}
