// Enrolar TOTP - devuelve el QR
export async function enrolarTotp(): Promise<{
  ok: boolean;
  factorId?: string;
  qrCode?: string;
  secret?: string;
  mensaje?: string;
}> {
  const { supabase, usuario } = await exigirSesion();
  
  const { data, error } = await supabase.auth.mfa.enroll({
    factorType: 'totp',
    friendlyName: 'Psicogestión',
  });

  if (error || !data) {
    console.error('Error enrolling TOTP:', error);
    return { ok: false, mensaje: t('primeraVez.totp.errorInterno') };
  }

  return {
    ok: true,
    factorId: data.id,
    qrCode: `data:image/svg+xml;utf-8,${encodeURIComponent(data.totp.qr_code)}`,
    secret: data.totp.secret,
  };
}

// Retar TOTP - crear el challenge
export async function crearRetoTotp(factorId: string): Promise<{
  ok: boolean;
  challengeId?: string;
  mensaje?: string;
}> {
  const { supabase } = await exigirSesion();

  const { data, error } = await supabase.auth.mfa.challenge({ factorId });

  if (error || !data) {
    console.error('Error creating TOTP challenge:', error);
    return { ok: false, mensaje: t('primeraVez.totp.errorInterno') };
  }

  return {
    ok: true,
    challengeId: data.id,
  };
}

// Verificar TOTP
export async function verificarTotp(
  factorId: string,
  challengeId: string,
  code: string
): Promise<EstadoFormulario> {
  const { supabase } = await exigirSesion();

  const { error } = await supabase.auth.mfa.verify({
    factorId,
    challengeId,
    code,
  });

  if (error) {
    console.error('TOTP verification error:', error);
    return {
      ok: false,
      mensaje: t('primeraVez.totp.errorVerificacion'),
    };
  }

  return { ok: true };
}

export async function accionFijarPin(datos: unknown): Promise<EstadoFormulario> {
  const sesion = await exigirSesion();
  
  const resultado = esquemaPin.safeParse(datos);
  
  if (!resultado.ok) {
    const errores = z.flattenError(resultado.error).fieldErrors;
    return { ok: false, errores };
  }

  const { pin } = resultado.data;
  
  const guardado = await fijarPinHistoria(sesion, pin);
  
  if (!guardado.ok) {
    return { ok: false, mensaje: guardado.mensaje };
  }

  return { ok: true };
}

  // Determinar a qué paso ir

    redirect('/primera-vez');
  }
  
  if (estado?.requierePin && !estado.tienePin) {
    redirect('/primera-vez');
  }
  
  // Si llegó aquí, algo extraño, redirigir a primera-vez
  redirect('/primera-vez');
}

  factorId: string;
  secret: string;
  qrCode: string;
  alVerificar: () => void;
}

type Estado = typeof ESTADO_INICIAL;

export function FormularioTotp({ factorId, secret, qrCode, alVerificar }: FormularioTotpProps) {
  const [desafioId, setDesafioId] = useState<string | null>(null);
  const [mostrarSecret, setMostrarSecret] = useState(false);
  const [estado, setEstado] = useState<Estado>(ESTADO_INICIAL);
  const [enviando, setEnviando] = useState(false);

  const {
    register,
    handleSubmit,
    setError,
    formState: { errors },
  } = useForm<DatosVerificacionTotp>({
    resolver: zodResolver(esquemaVerificacionTotp),
    defaultValues: {
      factorId,
      codigo: '',
    },

    setEnviando(true);
    setEstado(ESTADO_INICIAL);

    try {
      // Crear desafío si no existe
      let challengeId = desafioId;
      if (!challengeId) {
        const reto = await crearRetoTotp(factorId);
        if (!reto.ok) {
          setEstado({ mensaje: reto.mensaje });
          setEnviando(false);
          return;

        challengeId = reto.challengeId;
        setDesafioId(challengeId);
      }

      const resultado = await verificarTotp(factorId, challengeId, datos.codigo);

        if (resultado.errores) {
          aplicarErroresDelServidor(resultado.errores, setError);
          setEstado({ mensaje: resultado.mensaje });
        } else {
          setEstado({ mensaje: resultado.mensaje });
        }
        setDesafioId(null); // Resetear desafío para reintentar
      } else {
        alVerificar();
      }
    } catch (e) {
      console.error('Error en verificación TOTP:', e);
      setEstado({ mensaje: t('primeraVez.totp.errorInterno') });
    } finally {
      setEnviando(false);
    }
  });

  return (
    <div className="space-y-6">
      <div className="flex flex-col items-center gap-4">
        <img
          src={qrCode}
          alt="Código QR de autenticación"
          className="w-48 h-48"
        />
        
        <div className="text-center text-sm text-gray-600">
          <p>{t('primeraVez.totp.descripcion')}</p>
        </div>
        
        <button
          type="button"
          onClick={() => setMostrarSecret(!mostrarSecret)}
          className="text-sm text-blue-600 hover:underline"
        >
          {mostrarSecret ? t('primeraVez.totp.oManual').split(':')[0] : t('primeraVez.totp.oManual')}
        </button>
        
        {mostrarSecret && (
          <code className="bg-gray-100 px-3 py-2 rounded text-sm font-mono">
            {secret}
          </code>
        )}
      </div>

      <form onSubmit={enviar} className="space-y-4">
        <Field
          id="codigo"
          etiqueta={t('primeraVez.totp.codigoPlaceholder')}
          errores={errors.codigo?.message ? [errors.codigo.message] : undefined}
          {...register('codigo')}
        />
        <input type="hidden" {...register('factorId')} />

type Paso = 'totp' | 'codigos' | 'pin' | 'completado';

function determinarPaso(estado: Awaited<ReturnType<typeof estadoDeCuenta>> | null, aal: string | null): Paso {
  // Si no tiene AAL2, necesita TOTP
  if (aal !== 'aal2') {
    return 'totp';
  }
  
  // Si tiene AAL2 pero no tiene PIN y lo requiere
  if (estado?.requierePin && !estado.tienePin) {
    return 'pin';
  }
  
  // Todo completado
  return 'completado';
}

export async function ContenidoPrimeraVez() {
  const { supabase } = await exigirSesion();
  
  // Obtener nivel de autenticación
  const { data: aalData } = await supabase.auth.mfa.getAuthenticatorAssuranceLevel();
  const aal = aalData?.currentLevel ?? null;
  
  // Obtener estado de la cuenta
  const { supabase: clienteAdmin, usuario } = await exigirSesion();
  const estado = await estadoDeCuenta({ supabase: clienteAdmin, usuario });
  
  // Determinar paso
  const paso = determinarPaso(estado, aal);
  
  // Si ya está completado, redirigir
  if (paso === 'completado') {
    redirect('/pacientes');
  }
  
  // Obtener códigos si es ese paso
  let codigos: string[] | null = null;
  if (paso === 'codigos') {
    const resultado = await obtenerCodigosRecuperacion();
    if (resultado.ok && resultado.codigos) {
      codigos = resultado.codigos;
    }
  }
  
  // Renderizar el paso correspondiente
  return (
    <div className="bg-white rounded-lg shadow-md p-6">
      <h1 className="text-2xl font-bold text-gray-900 mb-2">
        {t('primeraVez.titulo')}
      </h1>
      <p className="text-gray-600 mb-6">
        {t('primeraVez.subtitulo')}
      </p>
      
      {/* Indicador de pasos */}
      <div className="flex gap-2 mb-8">
        {(['totp', 'codigos', 'pin'] as const).map((p, i) => (
          <div
            key={p}
            className={`h-2 flex-1 rounded ${
              p === paso ? 'bg-blue-600' :
              ['totp', 'codigos', 'pin'].indexOf(paso) > i ? 'bg-green-500' : 'bg-gray-200'
            }`}
          />
        ))}
      </div>
      
      {paso === 'totp' && <PasoTotp />}
      {paso === 'codigos' && <PasoCodigos codigos={codigos} />}
      { paso === 'pin' && <PasoPin />}
    </div>
  );
}

import { useState, useEffect } from 'react';
import { enrolarTotp, verificarTotp } from './acciones';
import { FormularioTotp } from './formulario-totp';

export function PasoTotp() {
  const [cargando, setCargando] = useState(true);
  const [factorId, setFactorId] = useState<string | null>(null);
  const [qrCode, setQrCode] = useState<string | null>(null);
  const [secret, setSecret] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function inicializar() {
      const resultado = await enrolarTotp();
      if (resultado.ok && resultado.factorId && resultado.qrCode && resultado.secret) {
        setFactorId(resultado.factorId);
        setQrCode(resultado.qrCode);
        setSecret(resultado.secret);
      } else {
        setError(resultado.mensaje ?? 'Error desconocido');
      }
      setCargando(false);
    }
    inicializar();
  }, []);

  const handleVerificar = () => {
    // Recargar la página para mostrar el siguiente paso
    window.location.reload();
  };

  if (cargando) {
    return <div className="text-center py-8">Cargando...</div>;
  }

  if (error) {
    return <div className="text-center text-red-600 py-8">{error}</div>;
  }

  if (!factorId || !qrCode || !secret) {
    return <div className="text-center py-8">Error al cargar el código QR</div>;
  }

  return (
    <FormularioTotp
      factorId={factorId}
      secret={secret}
      qrCode={qrCode}
      alVerificar={handleVerificar}
    />
  );
}

  const [factorId, setFactorId] = useState<string | null>(null);
  const [qrCode, setQrCode] = useState<string | null>(null);

      if (resultado.ok && resultado.factorId && resultado.qrCode && resultado.secret) {
        setFactorId(resultado.factorId);
        setQrCode(resultado.qrCode);
        setSecret(resultado.secret);
        setSubPaso('totp');

        setSubPaso('error');

      setSubPaso('error');
    }

    // Recargar para verificar si necesita PIN
    window.location.reload();

      </div>
    );
