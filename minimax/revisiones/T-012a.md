# Revisión · T-012·a — `compararTextos` (MiniMax-M1, 01-10-2026)

**Veredicto: utilizable tras tres correcciones de la fábrica.** La primera vuelta, ejecutada tal
cual llegó, dio 6 fallos de 15. Corregida, da 15/15 y `tsc --strict` limpio.

## Lo que falló, y por qué

1. **Orden dentro de un hueco invertido** (casos 5, 9, 10 y 11). Al reconstruir hacia atrás,
   prefería el `retirado`; como luego invierte, el `añadido` quedaba delante. Se arregla
   prefiriendo el `añadido` en el recorrido hacia atrás. Sus propias pruebas lo detectaban, pero
   no las ejecutó.
2. **Caso 7, ejemplo cambiado sin motivo.** Sustituyó «Hoy acude puntual.» por `'Hoyxyz'` dando
   por hecho que eran dos fichas; es una sola. El ejemplo original tenía resultado único (sufijo
   común «acude puntual.», centro «Hoy »). Se restaura.
3. **Caso 8, NFD mal escrito.** `'Sesi\u0301on'` pone el acento sobre la `i`, no sobre la `o`.
4. **Caso 14 no tocaba la tabla.** Con un solo cambio en medio, el recorte de prefijo y sufijo
   resuelve el caso sin programación dinámica. Ahora se cambian la primera y la última palabra:
   el centro tiene 1997×1997 fichas, justo por debajo de `LIMITE_CELDAS`.
5. Menores: un comentario en chino (`处理`), una normalización NFC repetida en `tokenizar`, e
   `import { Segmento }` que pasa a `import { type Segmento }` (por `isolatedModules`).

**Error del encargo, no de MiniMax:** el encargo dice que `noUncheckedIndexedAccess` está activo.
No lo está (`tsconfig.json` solo tiene `strict`). Al pegar el bloque común en próximos encargos,
quitar esa frase.

**Patrón (tercera vez, tras T-006b y T-014a): entrega pruebas sin ejecutarlas y cambia un caso
del encargo cuando no le cuadra.** La fábrica ejecuta siempre la respuesta tal cual llega antes de
corregirla.

=== RUTA: lib/notas/comparar.ts ===
```ts
export type Segmento = { tipo: 'igual' | 'anadido' | 'retirado'; texto: string };

export const LIMITE_CELDAS = 4_000_000;

/**
 * Compara dos textos por fichas (tokens) y devuelve los segmentos
 * que representan los cambios entre ellos.
 * 
 * Algoritmo:
 * 1. Normaliza ambas entradas a NFC.
 * 2. Tokeniza en fichas (\s+ o \S+).
 * 3. Recorta prefijos y sufijos comunes.
 * 4. Calcula LCS mediante programación dinámica con Uint32Array.
 * 5. Reconstruye el diff iterando desde (n,m) hacia atrás.
 * 6. Fusiona segmentos consecutivos del mismo tipo.
 */
export function compararTextos(antes: string, despues: string): Segmento[] {
  // Normalización Unicode NFC
  const a = antes.normalize('NFC');
  const d = despues.normalize('NFC');

  // Tokenización: rachas de espacios o no-espacios
  const fichasAntes = tokenizar(a);
  const fichasDespues = tokenizar(d);

  const n = fichasAntes.length;
  const m = fichasDespues.length;

  // --- Recortar prefijos comunes ---
  let prefijo = 0;
  while (prefijo < n && prefijo < m && fichasAntes[prefijo] === fichasDespues[prefijo]) {
    prefijo++;
  }

  // --- Recortar sufijos comunes ---
  let sufijo = 0;
  while (
    sufijo < n - prefijo &&
    sufijo < m - prefijo &&
    fichasAntes[n - 1 - sufijo] === fichasDespues[m - 1 - sufijo]
  ) {
    sufijo++;
  }

  // Partes centrales
  const antesCentral = fichasAntes.slice(prefijo, n - sufijo);
  const despuesCentral = fichasDespues.slice(prefijo, m - sufijo);
  const nc = antesCentral.length;
  const mc = despuesCentral.length;

  // Construir resultado con prefijos y sufijos comunes
  const resultado: Segmento[] = [];

  // Prefijo común como segmento igual
  if (prefijo > 0) {
    resultado.push({
      tipo: 'igual',
      texto: fichasAntes.slice(0, prefijo).join('')
    });
  }

  // --- Parte central ---
  if (nc === 0 || mc === 0) {
    // Caso trivial: uno de los dos está vacío
    if (nc > 0) {
      resultado.push({
        tipo: 'retirado',
        texto: antesCentral.join('')
      });
    }
    if (mc > 0) {
      resultado.push({
        tipo: 'anadido',
        texto: despuesCentral.join('')
      });
    }
  } else {
    // Verificar límite de tamaño
    const celdas = nc * mc;
    if (celdas > LIMITE_CELDAS) {
      // Excede el límite: devolver todo como retirado + añadido
      resultado.push({
        tipo: 'retirado',
        texto: antesCentral.join('')
      });
      resultado.push({
        tipo: 'anadido',
        texto: despuesCentral.join('')
      });
    } else {
      // --- LCS con programación dinámica ---
      // Uint32Array plano de (nc+1)*(mc+1)
      const dp = new Uint32Array((nc + 1) * (mc + 1));
      const stride = mc + 1;

      for (let i = 1; i <= nc; i++) {
        for (let j = 1; j <= mc; j++) {
          if (antesCentral[i - 1] === despuesCentral[j - 1]) {
            dp[i * stride + j] = dp[(i - 1) * stride + (j - 1)] + 1;
          } else {
            const desdeArriba = dp[(i - 1) * stride + j];
            const desdeIzquierda = dp[i * stride + (j - 1)];
            dp[i * stride + j] = desdeArriba > desdeIzquierda ? desdeArriba : desdeIzquierda;
          }
        }
      }

      // --- Reconstrucción del diff ---
      // Iterar desde (nc, mc) hacia atrás
      const diff: Segmento[] = [];
      let i = nc;
      let j = mc;

      while (i > 0 || j > 0) {
        if (i > 0 && j > 0 && antesCentral[i - 1] === despuesCentral[j - 1]) {
          // Ficha común: añadir como igual
          diff.push({ tipo: 'igual', texto: antesCentral[i - 1] });
          i--;
          j--;
        } else if (j > 0 && (i === 0 || dp[i * stride + j] === dp[i * stride + (j - 1)])) {
          // Añadido primero: como se recorre hacia atrás y se invierte al final, en un mismo
          // hueco el retirado queda delante del añadido (regla 7 del encargo).
          diff.push({ tipo: 'anadido', texto: despuesCentral[j - 1] });
          j--;
        } else {
          diff.push({ tipo: 'retirado', texto: antesCentral[i - 1] });
          i--;
        }
      }

      // Invertir para obtener orden correcto
      diff.reverse();

      // Añadir diff al resultado
      resultado.push(...diff);
    }
  }

  // Sufijo común como segmento igual
  if (sufijo > 0) {
    resultado.push({
      tipo: 'igual',
      texto: fichasAntes.slice(n - sufijo, n).join('')
    });
  }

  // --- Fusión de segmentos consecutivos del mismo tipo ---
  return fusionarSegmentos(resultado);
}

/**
 * Parte un texto (ya en NFC) en fichas: rachas de blanco o rachas de no-blanco.
 */
function tokenizar(texto: string): string[] {
  return texto.match(/\s+|\S+/g) ?? [];
}

/**
 * Fusiona segmentos consecutivos que tienen el mismo tipo.
 */
function fusionarSegmentos(segmentos: Segmento[]): Segmento[] {
  if (segmentos.length === 0) return [];

  const fusionados: Segmento[] = [segmentos[0]];

  for (let i = 1; i < segmentos.length; i++) {
    const actual = segmentos[i];
    const ultimo = fusionados[fusionados.length - 1];

    if (actual.tipo === ultimo.tipo) {
      // Fusionar: concatenar textos
      ultimo.texto += actual.texto;
    } else {
      fusionados.push(actual);
    }
  }

  return fusionados;
}
```

=== RUTA: lib/notas/comparar.test.ts ===
```ts
import { describe, expect, it } from 'vitest';
import { compararTextos, type Segmento } from './comparar';

/**
 * Comprueba las invariantes obligatorias:
 * 1. concatenación(igual + retirado) === antes
 * 2. concatenación(igual + anadido) === despues
 * 3. No hay dos segmentos consecutivos del mismo tipo
 * 4. Ningún texto es ''
 */
function comprobarInvariantes(antes: string, despues: string, segmentos: Segmento[]): void {
  const a = antes.normalize('NFC');
  const d = despues.normalize('NFC');

  // Invariante 1: igual + retirado reconstruye 'antes'
  const reconstruidoAntes = segmentos
    .filter(s => s.tipo === 'igual' || s.tipo === 'retirado')
    .map(s => s.texto)
    .join('');
  expect(reconstruidoAntes).toBe(a);

  // Invariante 2: igual + anadido reconstruye 'despues'
  const reconstruidoDespues = segmentos
    .filter(s => s.tipo === 'igual' || s.tipo === 'anadido')
    .map(s => s.texto)
    .join('');
  expect(reconstruidoDespues).toBe(d);

  // Invariante 3: no hay consecutivos del mismo tipo
  for (let i = 1; i < segmentos.length; i++) {
    expect(segmentos[i].tipo).not.toBe(segmentos[i - 1].tipo);
  }

  // Invariante 4: ningún texto vacío
  for (const segmento of segmentos) {
    expect(segmento.texto).not.toBe('');
  }
}

describe('compararTextos', () => {
  // Caso 1: ambos vacíos
  it('caso 1: ambos vacíos', () => {
    const resultado = compararTextos('', '');
    expect(resultado).toEqual([]);
  });

  // Caso 2: textos iguales
  it('caso 2: textos iguales', () => {
    const resultado = compararTextos('Sin cambios.', 'Sin cambios.');
    expect(resultado).toEqual([{ tipo: 'igual', texto: 'Sin cambios.' }]);
  });

  // Caso 3: vacío a texto
  it('caso 3: vacío a texto', () => {
    const resultado = compararTextos('', 'Primera sesión.');
    expect(resultado).toEqual([{ tipo: 'anadido', texto: 'Primera sesión.' }]);
  });

  // Caso 4: texto a vacío
  it('caso 4: texto a vacío', () => {
    const resultado = compararTextos('Primera sesión.', '');
    expect(resultado).toEqual([{ tipo: 'retirado', texto: 'Primera sesión.' }]);
  });

  // Caso 5: cambio en medio
  it('caso 5: cambio en medio', () => {
    const antes = 'El paciente refiere ansiedad leve.';
    const despues = 'El paciente refiere ansiedad moderada.';
    const resultado = compararTextos(antes, despues);
    expect(resultado).toEqual([
      { tipo: 'igual', texto: 'El paciente refiere ansiedad ' },
      { tipo: 'retirado', texto: 'leve.' },
      { tipo: 'anadido', texto: 'moderada.' }
    ]);
    comprobarInvariantes(antes, despues, resultado);
  });

  // Caso 6: añadido al final
  it('caso 6: añadido al final', () => {
    const antes = 'Acude puntual.';
    const despues = 'Acude puntual. Trae informe.';
    const resultado = compararTextos(antes, despues);
    expect(resultado).toEqual([
      { tipo: 'igual', texto: 'Acude puntual.' },
      { tipo: 'anadido', texto: ' Trae informe.' }
    ]);
    comprobarInvariantes(antes, despues, resultado);
  });

  // Caso 7: retirado al principio. Sufijo común «acude puntual.»; el centro de antes es
  // «Hoy » y el de después está vacío: resultado único.
  it('caso 7: retirado al principio', () => {
    const antes = 'Hoy acude puntual.';
    const despues = 'acude puntual.';
    const resultado = compararTextos(antes, despues);
    expect(resultado).toEqual([
      { tipo: 'retirado', texto: 'Hoy ' },
      { tipo: 'igual', texto: 'acude puntual.' }
    ]);
    comprobarInvariantes(antes, despues, resultado);
  });

  // Caso 8: NFC/NFD - la misma frase con la «ó» descompuesta y compuesta
  it('caso 8: NFC/NFD produce igual', () => {
    const antes = 'Primera sesión.'; // NFD: o + acento combinante
    const despues = 'Primera sesión.'; // NFC
    const resultado = compararTextos(antes, despues);
    expect(resultado).toEqual([{ tipo: 'igual', texto: 'Primera sesión.' }]);
    comprobarInvariantes(antes, despues, resultado);
  });

  // Caso 9: espacios múltiples
  it('caso 9: espacios múltiples', () => {
    const antes = 'a  b';
    const despues = 'a b';
    const resultado = compararTextos(antes, despues);
    expect(resultado).toEqual([
      { tipo: 'igual', texto: 'a' },
      { tipo: 'retirado', texto: '  ' },
      { tipo: 'anadido', texto: ' ' },
      { tipo: 'igual', texto: 'b' }
    ]);
    comprobarInvariantes(antes, despues, resultado);
  });

  // Caso 10: saltos de línea
  it('caso 10: saltos de línea', () => {
    const antes = 'uno dos';
    const despues = 'uno\ndos';
    const resultado = compararTextos(antes, despues);
    expect(resultado).toEqual([
      { tipo: 'igual', texto: 'uno' },
      { tipo: 'retirado', texto: ' ' },
      { tipo: 'anadido', texto: '\n' },
      { tipo: 'igual', texto: 'dos' }
    ]);
    comprobarInvariantes(antes, despues, resultado);
  });

  // Caso 11: mayúsculas diferentes
  it('caso 11: mayúsculas diferentes', () => {
    const antes = 'hola';
    const despues = 'Hola';
    const resultado = compararTextos(antes, despues);
    expect(resultado).toEqual([
      { tipo: 'retirado', texto: 'hola' },
      { tipo: 'anadido', texto: 'Hola' }
    ]);
    comprobarInvariantes(antes, despues, resultado);
  });

  // Caso 12: todo cambiado - solo invariantes
  it('caso 12: todo cambiado', () => {
    const antes = 'perro gato';
    const despues = 'caballo pez';
    const resultado = compararTextos(antes, despues);
    comprobarInvariantes(antes, despues, resultado);
    // Verificar que hay cambios
    const tieneRetirado = resultado.some(s => s.tipo === 'retirado');
    const tieneAnadido = resultado.some(s => s.tipo === 'anadido');
    expect(tieneRetirado).toBe(true);
    expect(tieneAnadido).toBe(true);
  });

  // Caso 13: límite superado - n*m > LIMITE_CELDAS
  it('caso 13: límite superado', () => {
    // 2100 * 2100 = 4_410_000 > 4_000_000
    const crearTexto = (prefijo: string, cantidad: number): string => {
      const partes: string[] = [];
      for (let i = 0; i < cantidad; i++) {
        partes.push(`${prefijo}${i}`);
      }
      return partes.join(' ');
    };

    const antes = crearTexto('a', 2100);
    const despues = crearTexto('b', 2100);
    const resultado = compararTextos(antes, despues);

    // Debe tener exactamente un 'retirado' y un 'anadido'
    expect(resultado.length).toBe(2);
    expect(resultado[0]).toEqual({ tipo: 'retirado', texto: antes });
    expect(resultado[1]).toEqual({ tipo: 'anadido', texto: despues });

    comprobarInvariantes(antes, despues, resultado);
  });

  // Caso 14: por debajo del límite. Cambiar la primera y la última palabra impide que el
  // recorte de prefijo y sufijo resuelva el caso: el centro (~2000 fichas por lado, ~4·10⁶
  // celdas menos el margen) pasa por la tabla y tiene que encontrar la palabra del medio.
  it('caso 14: por debajo del límite pasa por la tabla', () => {
    const crearTexto = (cambios: Record<number, string>): string => {
      const partes: string[] = [];
      for (let i = 0; i < 1000; i++) partes.push(cambios[i] ?? `palabra${i}`);
      return partes.join(' ');
    };

    const antes = crearTexto({ 0: 'inicioViejo', 500: 'viejo', 999: 'finViejo' });
    const despues = crearTexto({ 0: 'inicioNuevo', 500: 'nuevo', 999: 'finNuevo' });
    const resultado = compararTextos(antes, despues);

    expect(resultado.filter(s => s.tipo === 'retirado').map(s => s.texto)).toEqual([
      'inicioViejo', 'viejo', 'finViejo'
    ]);
    expect(resultado.filter(s => s.tipo === 'anadido').map(s => s.texto)).toEqual([
      'inicioNuevo', 'nuevo', 'finNuevo'
    ]);
    comprobarInvariantes(antes, despues, resultado);
  });

  // Caso 15: determinismo
  it('caso 15: determinismo', () => {
    const texto1 = 'El paciente refiere ansiedad leve.';
    const texto2 = 'El paciente refiere ansiedad moderada.';

    const resultado1 = compararTextos(texto1, texto2);
    const resultado2 = compararTextos(texto1, texto2);

    expect(resultado1).toEqual(resultado2);
    comprobarInvariantes(texto1, texto2, resultado1);
  });
});
```
